cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" MDAMP_VERSION)
string(STRIP "${MDAMP_VERSION}" MDAMP_VERSION)
if(NOT MDAMP_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must contain major.minor.patch")
endif()
project(MdAmp VERSION ${MDAMP_VERSION} LANGUAGES CXX RC)
configure_file(src/version.h.in generated/mdamp_version.h @ONLY)
configure_file(src/MdAmp.rc.in generated/MdAmp.rc @ONLY)

if(NOT WIN32)
    message(FATAL_ERROR "MdAmp currently targets Windows only.")
endif()

add_library(mdamp_options INTERFACE)
target_compile_features(mdamp_options INTERFACE cxx_std_17)
target_compile_definitions(mdamp_options INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(mdamp_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_include_directories(mdamp_options INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
    target_compile_options(mdamp_options INTERFACE /W4 /WX /utf-8 /permissive-)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(mdamp_audio STATIC src/audio_engine.cpp)
target_link_libraries(mdamp_audio PUBLIC mdamp_options ole32)
add_library(mdamp_storage STATIC src/storage.cpp)
target_link_libraries(mdamp_storage PUBLIC mdamp_options)
add_library(MdAmpStereo SHARED plugins/stereo.cpp)
target_link_libraries(MdAmpStereo PRIVATE mdamp_options)
set_target_properties(MdAmpStereo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/plugins")
add_executable(MdAmp WIN32 src/main.cpp src/modern_ui.cpp src/MdAmp.manifest "${CMAKE_CURRENT_BINARY_DIR}/generated/MdAmp.rc")
target_link_libraries(MdAmp PRIVATE mdamp_options mdamp_audio mdamp_storage user32 gdi32 shell32 comctl32 comdlg32 ole32 uuid)
add_dependencies(MdAmp MdAmpStereo)
set_target_properties(MdAmp PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

include(CTest)
if(BUILD_TESTING)
    add_executable(mdamp_unit_tests tests/unit_tests.cpp)
    target_link_libraries(mdamp_unit_tests PRIVATE mdamp_options)
    add_test(NAME mdamp.unit COMMAND mdamp_unit_tests)
    add_test(NAME mdamp.window_smoke COMMAND MdAmp --smoke-test)
    set_tests_properties(mdamp.window_smoke PROPERTIES TIMEOUT 15)
    add_executable(mdamp_audio_tests tests/audio_tests.cpp)
    target_link_libraries(mdamp_audio_tests PRIVATE mdamp_audio)
    add_test(NAME mdamp.audio COMMAND mdamp_audio_tests)
    add_executable(mdamp_storage_tests tests/storage_tests.cpp)
    target_link_libraries(mdamp_storage_tests PRIVATE mdamp_storage)
    add_test(NAME mdamp.storage COMMAND mdamp_storage_tests)
    add_executable(mdamp_plugin_tests tests/plugin_tests.cpp)
    target_link_libraries(mdamp_plugin_tests PRIVATE mdamp_options)
    add_dependencies(mdamp_plugin_tests MdAmpStereo)
    add_test(NAME mdamp.plugin COMMAND mdamp_plugin_tests "$<TARGET_FILE:MdAmpStereo>")
    add_test(NAME mdamp.version COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/version_tests.ps1")
endif()

install(TARGETS MdAmp RUNTIME DESTINATION .)
install(TARGETS MdAmpStereo RUNTIME DESTINATION plugins)
